    function cleanDisplayProductName(rawProd) {
      if (!rawProd) return '';
      let p = String(rawProd).trim();
      p = p.replace(/\b\d{12,24}(-\d+)?\b/g, ' ');
      p = p.replace(/0\d{16,22}(-\d+)?/g, ' ');
      p = p.replace(/91\d{14,18}[A-Z0-9]?/gi, ' ');
      p = p.replace(/\b[A-Z]{3,4}\d{10,20}\b/gi, ' ');
      p = p.replace(/SB\s*PORK\s*A\/S/gi, ' ');
      p = p.replace(/[\u4e00-\u9fa5]+(有限公司|商貿公司|進出口公司|供應鏈|合作社|會社|貿易行|商行)/g, ' ');
      p = p.replace(/\b(s\.?a\.?|ltd|corp|inc|llc|gmbh|jsc|co\.|company|pty)\b/gi, ' ');
      p = p.replace(/^(俄羅斯|丹麥|厄瓜多|美國|日本|德國|澳洲|澳大利亞|韓國|巴西|阿曼|緬甸|泰國|越南|巴基斯坦)$/gi, '');

    function sanitizeCountryOnly(rawStr, rowPiecesText) {
      const cleanRaw = (rawStr || '').trim();

      if (STANDARD_COUNTRY_MAP[cleanRaw]) return STANDARD_COUNTRY_MAP[cleanRaw];
      if (STANDARD_COUNTRY_MAP[cleanRaw.toLowerCase()]) return STANDARD_COUNTRY_MAP[cleanRaw.toLowerCase()];

      if (cleanRaw && !ENTERPRISE_BLACKLIST_REGEX.test(cleanRaw)) {
        for (const [key, stdName] of Object.entries(STANDARD_COUNTRY_MAP)) {
          if (cleanRaw.toLowerCase().includes(key.toLowerCase())) return stdName;
        }
      }

      const searchContext = `${cleanRaw} ${rowPiecesText || ''}`;
      const sortedKeys = Object.keys(STANDARD_COUNTRY_MAP).sort((a, b) => b.length - a.length);
      for (const key of sortedKeys) {
        const stdName = STANDARD_COUNTRY_MAP[key];
        const re = new RegExp(`(^|[^a-zA-Z0-9\u4e00-\u9fa5])${key}([^a-zA-Z0-9\u4e00-\u9fa5]|$)`, 'i');
        if (re.test(searchContext)) return stdName;
      }

      if (/^[\u4e00-\u9fa5]{2,6}$/.test(cleanRaw) && !cleanRaw.includes('公司') && !cleanRaw.includes('企業')) {
        return STANDARD_COUNTRY_MAP[cleanRaw] || toTraditionalChinese(cleanRaw);
      }

      return '未標示國家';
    }

          if (activeFilters.month && !d.date.startsWith(activeFilters.month)) return false;
        }

        if (activeFilters.searchKeyword) {
          const rawKw = activeFilters.searchKeyword.toLowerCase().trim();
          const kwTrad = toTraditionalChinese(rawKw);
          const kwSimp = toSimplifiedChinese(rawKw);

          const matchCheck = (val) => {
            if (!val) return false;
            const strLower = String(val).toLowerCase();
            return strLower.includes(rawKw) || strLower.includes(kwTrad) || strLower.includes(kwSimp);
          };

          const matchKw = matchCheck(d.product) ||
                          matchCheck(d.country) ||
                          matchCheck(d.port) ||
                          matchCheck(d.detail) ||
                          matchCheck(d.tier3) ||
                          matchCheck(d.id) ||
                          matchCheck(d.declarationNo) ||
                          matchCheck(d.importer);
          if (!matchKw) return false;
        }

        return true;
      });
    }

    let treemapEventsBound = false;
    function initTreemapEvents(canvas) {
      if (treemapEventsBound) return;
      treemapEventsBound = true;
      const tooltip = document.getElementById('treemapTooltip');
      const ttTitle = document.getElementById('ttTitle');
      const ttDetail = document.getElementById('ttDetail');

      canvas.addEventListener('mousemove', (e) => {
        const rect = canvas.getBoundingClientRect();
        const mx = e.clientX - rect.left;
        const my = e.clientY - rect.top;
        const hit = treemapBoxes.find(b => mx >= b.x && mx <= b.x + b.w && my >= b.y && my <= b.y + b.h);
        if (hit) {
          tooltip.classList.remove('hidden');
          tooltip.style.left = `${Math.min(mx + 12, rect.width - 140)}px`;
          tooltip.style.top = `${Math.min(my + 12, rect.height - 60)}px`;
          ttTitle.textContent = hit.name;
          ttDetail.textContent = `${hit.count} 件 (${hit.pct}%)`;
        } else {
          tooltip.classList.add('hidden');
        }
      });

      canvas.addEventListener('mouseleave', () => {
        tooltip.classList.add('hidden');
      });

      canvas.addEventListener('click', (e) => {
        const rect = canvas.getBoundingClientRect();
        const mx = e.clientX - rect.left;
        const my = e.clientY - rect.top;
        const hit = treemapBoxes.find(b => mx >= b.x && mx <= b.x + b.w && my >= b.y && my <= b.y + b.h);
        if (hit) {
          if (hit.isOther) {
            isOriginDrilldown = true;
            renderOriginChart();
          } else {
            activeFilters.origin = (activeFilters.origin === hit.name) ? null : hit.name;
            refreshAll();
          }
        }
      });
    }

    function drawTreemap(canvas, items, totalCount) {
      initTreemapEvents(canvas);
      const rect = canvas.getBoundingClientRect();
      const dpr = window.devicePixelRatio || 1;
      const width = rect.width || 500;
      const height = rect.height || 290;
      canvas.width = width * dpr;
      canvas.height = height * dpr;
      const ctx = canvas.getContext('2d');
      ctx.scale(dpr, dpr);
      ctx.clearRect(0, 0, width, height);

      treemapBoxes = [];
      if (!items || items.length === 0) {
        ctx.fillStyle = '#64748b';
        ctx.font = '12px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText('無相符產地資料', width / 2, height / 2);
        return;
      }

      const colors = [
        '#0284c7', '#0d9488', '#059669', '#d97706', '#dc2626',
        '#7c3aed', '#db2777', '#2563eb', '#16a34a', '#ca8a04',
        '#e11d48', '#9333ea', '#4f46e5', '#0891b2', '#65a30d',
        '#c026d3', '#ea580c', '#334155', '#0f766e', '#854d0e'
      ];

      function partition(itemList, x, y, w, h) {
        if (itemList.length === 0) return;
        if (itemList.length === 1) {
          treemapBoxes.push({
            x, y, w, h,
            name: itemList[0].name,
            count: itemList[0].count,
            pct: ((itemList[0].count / totalCount) * 100).toFixed(1),
            isOther: itemList[0].isOther,
            color: itemList[0].isOther ? '#334155' : colors[treemapBoxes.length % colors.length]
          });
          return;
        }

        const sum = itemList.reduce((acc, it) => acc + it.count, 0) || 1;
        let half = 0;
        let splitIdx = 0;
        for (let i = 0; i < itemList.length; i++) {
          half += itemList[i].count;
          if (half >= sum / 2 || i === itemList.length - 2) {
            splitIdx = i + 1;
            break;
          }
        }
        if (splitIdx === 0) splitIdx = 1;

        const leftList = itemList.slice(0, splitIdx);
        const rightList = itemList.slice(splitIdx);
        const leftRatio = (itemList.slice(0, splitIdx).reduce((a, b) => a + b.count, 0)) / sum;

        if (w >= h) {
          const w1 = w * leftRatio;
          partition(leftList, x, y, w1, h);
          partition(rightList, x + w1, y, w - w1, h);
        } else {
          const h1 = h * leftRatio;
          partition(leftList, x, y, w, h1);
          partition(rightList, x, y + h1, w, h - h1);
        }
      }

      partition(items, 2, 2, width - 4, height - 4);

      treemapBoxes.forEach(b => {
        const isSelected = activeFilters.origin === b.name;
        ctx.fillStyle = isSelected ? '#38bdf8' : b.color;
        ctx.fillRect(b.x + 1, b.y + 1, Math.max(b.w - 2, 1), Math.max(b.h - 2, 1));

        ctx.strokeStyle = isSelected ? '#ffffff' : '#070b13';
        ctx.lineWidth = isSelected ? 2 : 1;
        ctx.strokeRect(b.x + 1, b.y + 1, Math.max(b.w - 2, 1), Math.max(b.h - 2, 1));

        if (b.w > 28 && b.h > 24) {
          ctx.fillStyle = '#ffffff';
          ctx.font = 'bold 11px sans-serif';
          ctx.textAlign = 'center';
          ctx.textBaseline = 'middle';
          let displayName = b.name;
          if (ctx.measureText(displayName).width > b.w - 6) {
            displayName = displayName.slice(0, Math.max(2, Math.floor((b.w - 8) / 12))) + '..';
          }
          ctx.fillText(displayName, b.x + b.w / 2, b.y + b.h / 2 - (b.h > 38 ? 7 : 0));

          if (b.h > 38) {
            ctx.fillStyle = '#cbd5e1';
            ctx.font = '10px monospace';
            ctx.fillText(`${b.count}件`, b.x + b.w / 2, b.y + b.h / 2 + 8);
          }
        }
      });
    }

    function renderProductChart() {
      const data = getFilteredData('product');
      const counts = {};

      const badge = document.getElementById('productLevelBadge');
      const btnBack = document.getElementById('btnReturnProduct');
      const hint = document.getElementById('productFooterHint');
      const indicator = document.getElementById('productViewIndicator');

      let currentColor = '#10b981';

      if (productHierarchyLevel === 1) {
        badge.textContent = "Tier 1: 16大標準品類";
        btnBack.classList.add('hidden');
        hint.textContent = "點擊品類下鑽次分類；生鮮原肉（雞/豬/牛/羊/魚）單獨一類無須次分";
        indicator.textContent = "1/3 階 (品類)";

        data.forEach(d => {
          const simpleName = formatCategorySimple(d.category);
          counts[simpleName] = (counts[simpleName] || 0) + 1;
        });
      } else if (productHierarchyLevel === 2) {
        badge.textContent = `Tier 2: 【${formatCategorySimple(activeFilters.productCat)}】次分類`;
        btnBack.classList.remove('hidden');
        hint.textContent = "點擊次分類長條柱下鑽至具體申報品名 Top 10";
        indicator.textContent = "2/3 階 (次分類)";
        currentColor = '#06b6d4';

        data.filter(d => d.category === activeFilters.productCat).forEach(d => {
          const sub = d.subCategory || '其餘加工品';
          counts[sub] = (counts[sub] || 0) + 1;
        });
      } else {
        badge.textContent = `Tier 3: 【${activeFilters.productSub || formatCategorySimple(activeFilters.productCat)}】品名細項 Top 10`;
        btnBack.classList.remove('hidden');
        hint.textContent = "點擊具體品名細項長條柱直接鎖定全域篩選";
        indicator.textContent = "3/3 階 (具體品名)";
        currentColor = '#f59e0b';

        data.filter(d => {
          if (activeFilters.productSub) return d.subCategory === activeFilters.productSub;
          if (activeFilters.productCat) return d.category === activeFilters.productCat;
          return true;
        }).forEach(d => {
          const cleanP = cleanDisplayProductName(d.product) || formatCategorySimple(d.category);
          counts[cleanP] = (counts[cleanP] || 0) + 1;
        });
      }

      let sorted = Object.entries(counts).sort((a,b) => b[1] - a[1]);
      if (productHierarchyLevel === 3) sorted = sorted.slice(0, 10);

      const labels = sorted.map(s => s[0]);
      const values = sorted.map(s => s[1]);

      if (chartProductInstance) chartProductInstance.destroy();
      const ctx = document.getElementById('chartProductCanvas').getContext('2d');
      chartProductInstance = new Chart(ctx, {
        type: 'bar',
        data: {
          labels,
          datasets: [{
            data: values,
            backgroundColor: currentColor,
            hoverBackgroundColor: '#34d399',
            borderRadius: 6
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: { legend: { display: false } },
          scales: {
            y: { grid: { color: 'rgba(51, 65, 85, 0.4)' }, ticks: { color: '#94a3b8' } },
            x: {
              grid: { display: false },
              ticks: {
                color: '#cbd5e1',
                callback: function(val, index) {
                  const lbl = this.getLabelForValue(val) || '';
                  return lbl.length > 8 ? lbl.slice(0,7) + '..' : lbl;
                }
              }
            }
          },
          onClick: (evt, elements) => {
            if (elements.length > 0) {
              const idx = elements[0].index;
              const selected = labels[idx];

              if (productHierarchyLevel === 1) {
                const targetFullCat = PRODUCT_CATEGORIES_16.find(c => formatCategorySimple(c) === selected) || selected;
                if (RAW_MEAT_CATEGORIES.some(rm => formatCategorySimple(rm) === selected)) {
                  activeFilters.productCat = (activeFilters.productCat === targetFullCat) ? null : targetFullCat;
                } else {
                  activeFilters.productCat = targetFullCat;
                  productHierarchyLevel = 2;
                }
              } else if (productHierarchyLevel === 2) {
                activeFilters.productSub = selected;
                productHierarchyLevel = 3;
              } else {
                activeFilters.productDetail = (activeFilters.productDetail === selected) ? null : selected;
              }
              setTimeout(() => refreshAll(), 0);
            }
          }
        }
      });
    }

    function renderTable() {
      const data = getFilteredData();
      const tbody = document.getElementById('tableBody');
      tbody.innerHTML = '';

      const total = data.length;
      const totalPages = Math.ceil(total / pageSize) || 1;
      if (currentPage > totalPages) currentPage = totalPages;

      const startIndex = (currentPage - 1) * pageSize;
      const pageData = data.slice(startIndex, startIndex + pageSize);

      document.getElementById('tableInfoText').textContent =
        total > 0 ? `顯示第 ${startIndex + 1} 至 ${Math.min(startIndex + pageSize, total)} 筆，共 ${total} 筆` : `無相符紀錄`;
      document.getElementById('pageIndicator').textContent = `${currentPage} / ${totalPages}`;
      document.getElementById('btnPrevPage').disabled = currentPage <= 1;
      document.getElementById('btnNextPage').disabled = currentPage >= totalPages;

    function exportFilteredData() {
      const data = getFilteredData();
      if (data.length === 0) return;

      const headers = ["通報日期", "官方案號", "查驗口岸", "原產地/國別", "產品名稱", "16大標準品類", "次分類", "危害大類", "危害中類", "具體違規事由", "處置結果", "進口商/申報者", "境外生產企業"];
      const rows = data.map(d => [
        `"${d.date}"`,
        `"${d.id}"`,
        `"${d.port}"`,
        `"${d.country}"`,
        `"${d.product}"`,
        `"${d.category}"`,
        `"${d.subCategory || ''}"`,
        `"${d.major}"`,
        `"${d.subHazard}"`,
        `"${(d.detail || '').replace(/"/g, '""')}"`,
        `"${d.decision}"`,
        `"${(d.importer || '').replace(/"/g, '""')}"`,
        `"${(d.overseasFirm || '').replace(/"/g, '""')}"`
      ]);

      const csvContent = "\uFEFF" + [headers.join(","), ...rows.map(r => r.join(","))].join("\n");
      const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = `China_Customs_GACC_Border_FoodSafety_${new Date().toISOString().slice(0,10)}.csv`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    }

    document.getElementById('fileInput').addEventListener('change', async (e) => {
      const files = Array.from(e.target.files);
      if (!files || files.length === 0) return;

      const banner = document.getElementById('pdfLoadingBanner');
      const progressText = document.getElementById('pdfLoadingText');
      const pageProgress = document.getElementById('pdfPageProgress');
      banner.classList.remove('hidden');

      let allExtractedRecords = [];

      for (let i = 0; i < files.length; i++) {
        const file = files[i];
        progressText.textContent = `[${i + 1}/${files.length}] 正在解析：${file.name}...`;

        try {
          let records = [];
          if (file.name.endsWith('.pdf')) {
            records = await parseSinglePdf(file, (curr, total) => {
              pageProgress.textContent = `頁次: ${curr}/${total}`;
            });
          } else if (file.name.endsWith('.csv')) {
            records = await parseSingleCsv(file);
          } else if (file.name.endsWith('.xlsx') || file.name.endsWith('.xls')) {
            records = await parseSingleExcel(file);
          }

          if (records && records.length > 0) {
            allExtractedRecords.push(...records);
            loadedFileNames.add(file.name);
          }
        } catch (err) {
          console.error(`解析檔案 ${file.name} 失敗:`, err);
        }
      }

      banner.classList.add('hidden');

      if (allExtractedRecords.length > 0) {
        mergeNewRecords(allExtractedRecords);
      } else {
        showCustomAlert('未能從選取的檔案中解析出邊境查驗資料，請確認檔案格式是否相符。');
      }

      e.target.value = '';
    });

    function mergeNewRecords(newRecords) {
      if (!isUserDatasetActive) {
        masterData = [];
        isUserDatasetActive = true;
      }

      masterData = [...masterData, ...newRecords];
      updateFileBadgeUI();
      resetAllFilters();

      showCustomAlert(`✅ 成功匯入入庫！\n本次共解析讀取：${newRecords.length} 筆案件\n目前戰情總庫存：${masterData.length} 筆真實資料 (絕不遺漏任何紀錄)`);
    }

    function updateFileBadgeUI() {
      const badge = document.getElementById('importedFilesBadge');
      if (loadedFileNames.size > 0) {
        badge.classList.remove('hidden');
        badge.textContent = `已載入 ${loadedFileNames.size} 個檔案: ` + Array.from(loadedFileNames).join(', ');
      } else {
        badge.classList.add('hidden');
      }
    }

    function clearAllDataToFresh() {
      masterData = [];
      loadedFileNames.clear();
      isUserDatasetActive = true;
      updateFileBadgeUI();
      resetAllFilters();
      showCustomAlert('已清空當前資料庫。現在可以開始匯入您的第一批 PDF 或 Excel 檔案！');
    }

    async function parseSinglePdf(file, pageProgressCallback) {

      const loadingTask = pdfjsLib.getDocument({ data: arrayBuffer });
      const pdf = await loadingTask.promise;
      const totalPages = pdf.numPages;

      let extractedRecords = [];
      let dateGuessed = '2026/08/15';

      const dateMatch = file.name.match(/(202\d)[-_年]?([01]?\d)/);
      if (dateMatch) {
        const yr = dateMatch[1];
        const mo = String(dateMatch[2]).padStart(2, '0');
        dateGuessed = `${yr}/${mo}/15`;
      }

      const fileKey = file.name.replace(/\.[^/.]+$/, "");
      const VIOLATION_REGEX = /焦亚硫酸钠|动物疫病|疫病|呋喃西林|转基因|感官检验|包装不合格|包裝不合格|标签不合格|標籤不合格|未按要求提供|未获准入|货证不符|企业注册问题|二氧化硫|挥发性盐基氮|酸价|过氧化值|储存温度|污秽腐败|霉变|镉检验值|铅检验值|重金属|三氯甲烷|甲醇|苯并|特丁硫磷|啶蜱脲|苹果酸|精氨酸|谷氨酰胺|日落黄|柠檬黄|亮蓝|大肠菌群|霉菌|菌落总数|有害生物|保质期|双咪苯脲|双味苯|退运|物种鉴定|不符合食品安全国家标准/i;
      const KNOWN_PORTS = ['北京', '成都', '大连', '福州', '广州', '南京', '南宁', '宁波', '青岛', '厦门', '上海', '深圳', '天津', '乌鲁木齐', '湛江', '重庆', '武汉', '海口', '拱北', '昆明', '哈尔滨', '长沙', '黄埔', '汕头'];

      for (let pageNum = 1; pageNum <= totalPages; pageNum++) {
        if (pageProgressCallback) pageProgressCallback(pageNum, totalPages);

        const page = await pdf.getPage(pageNum);
        const textContent = await page.getTextContent();
        const items = textContent.items;
        if (!items || items.length === 0) continue;

        const validItems = items.filter(it => {
          const str = (it.str || '').trim();

          if (str.includes('海关总署') || str.includes('未准入境的食品') || str.includes('郑重说明') || str.includes('鄭重說明')) return false;

        let headerY = -1;
        for (const it of validItems) {
          const s = it.str.trim();
          if (s.includes('序号') || s.includes('HS编码') || s.includes('报关单号') || s.includes('未准入境的事实')) {
            headerY = it.transform[5];
            break;
          }
        }

        const dataItems = headerY > 0 ? validItems.filter(it => it.transform[5] < headerY - 5) : validItems;
        if (dataItems.length === 0) continue;

        const minX = Math.min(...dataItems.map(it => it.transform[4]));
        const maxX = Math.max(...dataItems.map(it => it.transform[4] + (it.width || 0)));
        const pageW = maxX - minX || 800;

        const seqItems = [];
        dataItems.forEach(it => {
          const s = it.str.trim();
          const relX = (it.transform[4] - minX) / pageW;
          if (relX <= 0.12 && /^\d+$/.test(s)) {
            const num = parseInt(s, 10);
            if (num > 0 && num < 100000) {
              seqItems.push({ seq: num, y: it.transform[5], x: it.transform[4] });
            }
          }
        });

        seqItems.sort((a, b) => b.y - a.y);

        if (seqItems.length > 0) {
          for (let i = 0; i < seqItems.length; i++) {
            const curSeq = seqItems[i];
            const topY = curSeq.y + 12;
            const botY = (i < seqItems.length - 1) ? seqItems[i + 1].y + 12 : -9999;

            const rowPieces = dataItems.filter(it => it.transform[5] <= topY && it.transform[5] > botY);
            rowPieces.sort((a, b) => a.transform[4] - b.transform[4]);
            const rowTextCombined = rowPieces.map(p => p.str.trim()).join(' ');

            let hsCode = '';
            let declarationNo = '';
            let product = '';
            let rawCountryCandidate = '';
            let firmInfo = '';
            let firmRegNo = '';
            let importer = '';
            let importerRegNo = '';
            let weight = '';
            let detail = '';
            let port = '';

            const violationPieces = [];
            rowPieces.forEach(p => {
              const text = p.str.trim();
              if (VIOLATION_REGEX.test(text)) violationPieces.push(text);
            });
            if (violationPieces.length > 0) detail = violationPieces.join(' ');

            for (let k = rowPieces.length - 1; k >= 0; k--) {
              const t = rowPieces[k].str.trim();
              if (KNOWN_PORTS.some(kp => t.includes(kp)) || t.includes('海关') || t.includes('海關') || t.includes('口岸')) {
                port = t;
                break;
              }
            }

            const colTexts = Array.from({ length: 12 }, () => []);
            rowPieces.forEach(piece => {
              const rx = (piece.transform[4] - minX) / pageW;
              let colIdx = 0;
              if (rx < 0.04) colIdx = 0;
              else if (rx < 0.12) colIdx = 1;
              else if (rx < 0.22) colIdx = 2;
              else if (rx < 0.33) colIdx = 3;
              else if (rx < 0.42) colIdx = 4;
              else if (rx < 0.54) colIdx = 5;
              else if (rx < 0.63) colIdx = 6;
              else if (rx < 0.74) colIdx = 7;
              else if (rx < 0.83) colIdx = 8;
              else if (rx < 0.88) colIdx = 9;
              else if (rx < 0.96) colIdx = 10;
              else colIdx = 11;
              colTexts[colIdx].push(piece.str.trim());
            });

            const colStrs = colTexts.map(arr => arr.join(' ').trim());

            const seqNo = String(curSeq.seq);
            hsCode = colStrs[1] || '';
            declarationNo = colStrs[2] || '';
            product = colStrs[3] || '';
            rawCountryCandidate = colStrs[4] || '';
            firmInfo = colStrs[5] || '';
            firmRegNo = colStrs[6] || '';
            importer = colStrs[7] || '';
            importerRegNo = colStrs[8] || '';
            weight = colStrs[9] || '';

            if (!detail) detail = colStrs[10] || '不符合食品安全國家標準規定';

            const cleanCountry = sanitizeCountryOnly(rawCountryCandidate, rowTextCombined);

            let finalProduct = product;
            if (!finalProduct || finalProduct.length <= 2) {
              const prodCandidates = rowPieces.filter(p => {
                const text = p.str.trim();
                return text.length > 2 && !/^\d+$/.test(text) && !VIOLATION_REGEX.test(text) && !STANDARD_COUNTRY_MAP[text];
              });
              if (prodCandidates.length > 0) {
                finalProduct = prodCandidates.map(p => p.str.trim()).join('');
              } else {
                finalProduct = finalProduct || '進口食品';
              }
            }

            const catObj = autoCategorizeProduct(finalProduct);
            const hazObj = autoClassifyHazard(detail);

            const caseUid = `${fileKey}_${dateGuessed.replace(/\//g,'')}_seq${seqNo}_${declarationNo || Math.random().toString(36).substring(2,6)}`;

            extractedRecords.push({
              uid: caseUid,
              id: seqNo,
              hsCode,
              declarationNo,
              product: toTraditionalChinese(finalProduct),
              country: toTraditionalChinese(cleanCountry),
              firmInfo: toTraditionalChinese(firmInfo),
              firmRegNo,
              importer: toTraditionalChinese(importer),
              importerRegNo,
              weight,
              detail: toTraditionalChinese(detail),
              port: formatPortName(port),
              date: dateGuessed,
              category: toTraditionalChinese(catObj.category),
              subCategory: toTraditionalChinese(catObj.subCategory),
              major: toTraditionalChinese(hazObj.major),
              subHazard: toTraditionalChinese(hazObj.subHazard),
              tier3: toTraditionalChinese(hazObj.tier3),
              decision: '依法做退貨或銷毀處理'
            });
          }
        }
      }

      return extractedRecords;
    }

    async function parseSingleExcel(file) {
      const data = await file.arrayBuffer();
      const workbook = XLSX.read(data, { type: 'array' });
      const firstSheetName = workbook.SheetNames[0];
      const worksheet = workbook.Sheets[firstSheetName];
      const rows = XLSX.utils.sheet_to_json(worksheet, { header: 1, defval: '' });
      return map2DMatrixToRecords(rows, file.name);
    }

    async function parseSingleCsv(file) {
      return new Promise((resolve, reject) => {
        Papa.parse(file, {
          skipEmptyLines: false,
          complete: (results) => {
            resolve(map2DMatrixToRecords(results.data, file.name));
          },
          error: (err) => reject(err)
        });
      });
    }

    function map2DMatrixToRecords(rows, fileName) {
      if (!rows || rows.length === 0) return [];

      let defaultDate = '2026/08/15';
      const dateMatch = fileName.match(/(202\d)[-_年]?([01]?\d)/);
      if (dateMatch) {
        defaultDate = `${dateMatch[1]}/${String(dateMatch[2]).padStart(2, '0')}/15`;
      }

      const fileKey = fileName.replace(/\.[^/.]+$/, "");
      const headerKeywords = ['序号', 'hs编码', '报关单号', '产品名称', '品名', '产地', '生产企业', '注册号', '进口商', '重量', '未准入境', '进境口岸', '事实'];

      let headerRowIdx = -1;
      let maxMatches = 0;

      const scanLimit = Math.min(rows.length, 15);
      for (let r = 0; r < scanLimit; r++) {
        const row = rows[r];
        if (!Array.isArray(row)) continue;
        const rowStr = row.map(c => String(c || '').replace(/\s+/g, '').toLowerCase()).join(' ');
        let matches = 0;
        headerKeywords.forEach(kw => {
          if (rowStr.includes(kw)) matches++;
        });
        if (matches > maxMatches && matches >= 3) {
          maxMatches = matches;
          headerRowIdx = r;
        }
      }

      let headerHeaders = [];
      let startRow = 0;

      if (headerRowIdx !== -1) {
        headerHeaders = rows[headerRowIdx].map(c => String(c || '').replace(/\s+/g, '').toLowerCase());
        startRow = headerRowIdx + 1;
      } else {
        headerHeaders = (rows[0] || []).map(c => String(c || '').replace(/\s+/g, '').toLowerCase());
        startRow = 1;
      }

      const findExactColIdx = (exactNames, fallbackNames) => {
        for (const name of exactNames) {
          const idx = headerHeaders.findIndex(h => h === name.toLowerCase());
          if (idx !== -1) return idx;
        }
        for (const name of fallbackNames) {
          const idx = headerHeaders.findIndex(h => h.includes(name.toLowerCase()) && !h.includes('企业') && !h.includes('企業'));
          if (idx !== -1) return idx;
        }
        return -1;
      };

      const cId = findExactColIdx(['序号', '序號', '编号'], ['序号', '序號']);
      const cHs = findExactColIdx(['hs编码', 'hs編碼', 'hs', '税号'], ['hs']);
      const cDec = findExactColIdx(['报关单号', '報關單號', '单号'], ['报关', '報關']);
      const cProd = findExactColIdx(['产品名称', '產品名稱', '品名', '食品名称'], ['产品', '產品', '品名']);
      const cCountry = findExactColIdx(['产地', '產地', '原产国', '原產國', '出口国', '出口國', '国家'], ['产地', '產地', '原产国', '出口国']);
      const cFirm = findExactColIdx(['生产企业信息', '生產企業信息', '生产企业', '境外生产企业'], ['生产企业', '境外']);
      const cFirmReg = findExactColIdx(['生产企业注册号', '生產企業註冊號', '注册号'], ['注册号', '註冊號']);
      const cImp = findExactColIdx(['进口商信息', '進口商信息', '进口商', '申報單位'], ['进口商', '進口商']);
      const cImpReg = findExactColIdx(['进口商备案号', '進口商備案號', '备案号'], ['备案号', '備案號']);
      const cWeight = findExactColIdx(['重量', '净重'], ['重量', '净重']);
      const cDetail = findExactColIdx(['未准入境的事实', '未准入境事实', '未准入境原因', '不合格原因', '事实'], ['未准入境', '原因', '事实']);
      const cPort = findExactColIdx(['进境口岸', '進境口岸', '口岸', '关区'], ['口岸', '关区']);

      const records = [];
      const VIOLATION_REGEX = /焦亚硫酸钠|动物疫病|疫病|呋喃西林|转基因|感官检验|包装不合格|包裝不合格|标签不合格|標籤不合格|未按要求提供|未获准入|货证不符|企业注册问题|二氧化硫|挥发性盐基氮|酸价|过氧化值|储存温度|污秽腐败|霉变|镉检验值|铅检验值|重金属|三氯甲烷|甲醇|苯并|特丁硫磷|啶蜱脲|苹果酸|精氨酸|谷氨酰胺|日落黄|柠檬黄|亮蓝|大肠菌群|霉菌|菌落总数|有害生物|保质期|双咪苯脲|双味苯|退运|物种鉴定|不符合食品安全国家标准/i;

      for (let i = startRow; i < rows.length; i++) {
        const row = rows[i];
        if (!row || !Array.isArray(row) || row.length === 0) continue;

        const getColVal = (idx) => (idx !== -1 && row[idx] !== undefined && row[idx] !== null) ? String(row[idx]).trim() : '';

        const id = getColVal(cId) || String(records.length + 1);
        const hsCode = getColVal(cHs);
        const declarationNo = getColVal(cDec);
        const product = getColVal(cProd);
        const rawCountry = getColVal(cCountry);
        const firmInfo = getColVal(cFirm);
        const firmRegNo = getColVal(cFirmReg);
        const importer = getColVal(cImp);
        const importerRegNo = getColVal(cImpReg);
        const weight = getColVal(cWeight);
        let detail = getColVal(cDetail);
        const port = getColVal(cPort) || '北京海關';

        if (!product && !declarationNo && !detail && !firmInfo) continue;
        if (product.includes('产品名称') || detail.includes('未准入境的事实')) continue;

        if (!detail || !VIOLATION_REGEX.test(detail)) {
          for (let col = 0; col < row.length; col++) {
            const cellVal = String(row[col] || '').trim();
            if (VIOLATION_REGEX.test(cellVal)) {
              detail = cellVal;
              break;
            }
          }
        }
        if (!detail) detail = '不符合食品安全國家標準規定';

        detail = detail.replace(/[\r\n\t]+/g, ' ').replace(/\s+/g, ' ').replace(/^\d+[\.、\s]+/, '').trim();
        const rowCombinedStr = row.map(c => String(c || '')).join(' ');
        
        const cleanCountry = sanitizeCountryOnly(rawCountry, rowCombinedStr);
        const catObj = autoCategorizeProduct(product || '進口食品');
        const hazObj = autoClassifyHazard(detail);

        const caseUid = `${fileKey}_${defaultDate.replace(/\//g,'')}_seq${id}_${declarationNo || Math.random().toString(36).substring(2,6)}`;

        records.push({
          uid: caseUid,
          id,
          hsCode,
          declarationNo,
          product: toTraditionalChinese(product || '進口食品'),
          country: toTraditionalChinese(cleanCountry),
          firmInfo: toTraditionalChinese(firmInfo),
          firmRegNo,
          importer: toTraditionalChinese(importer),
          importerRegNo,
          weight,
          detail: toTraditionalChinese(detail),
          port: formatPortName(port),
          date: defaultDate,
          category: toTraditionalChinese(catObj.category),
          subCategory: toTraditionalChinese(catObj.subCategory),
          major: toTraditionalChinese(hazObj.major),
          subHazard: toTraditionalChinese(hazObj.subHazard),
          tier3: toTraditionalChinese(hazObj.tier3),
          decision: '依法做退貨或銷毀處理'
        });
      }

      return records;
    }
